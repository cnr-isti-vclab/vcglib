374 bool SnapPolyline(MeshType &poly)
375 {
376 tri::Allocator<MeshType>::CompactEveryVector(poly);
377 tri::UpdateTopology<MeshType>::VertexEdge(poly);
378 // Where each vertex would snap.
379 std::vector<FaceType *> face(poly.vert.size());
380 std::vector<CoordType> ip(poly.vert.size()), raw(poly.vert.size());
381 std::map<VertexType *, std::vector<size_t>> onVertex; // mesh vertex -> polyline vertices
382 for (size_t i = 0; i < poly.vert.size(); ++i)
383 {
384 face[i] = GetClosestFaceIP(poly.vert[i].cP(), raw[i]);
385 ip[i] = raw[i];
386 // A control point is not snapped; one already on a mesh vertex occupies it.
387 if (poly.vert[i].IsS()) RoundingSnap(ip[i], *face[i]);
388 else BarycentricSnap(ip[i], *face[i]);
389 if (VertexPointer v = FindVertexSnap(face[i], ip[i])) onVertex[v].push_back(i);
390 }
391 // A mesh vertex takes one polyline vertex: the nearest, with the ones joined to it by
392 // polyline edges (they will collapse onto it). The others stay off it, on an edge if
393 // they may snap there, so two strands passing by the same vertex are not merged.
394 int deniedCnt = 0;
395 for (auto &ov : onVertex)
396 {
397 if (ov.second.size() < 2) continue;
398 std::sort(ov.second.begin(), ov.second.end(), [&](size_t a, size_t b) {
399 if (poly.vert[a].IsS() != poly.vert[b].IsS()) return poly.vert[a].IsS(); // a control point first
400 return Distance(poly.vert[a].cP(), ov.first->cP()) < Distance(poly.vert[b].cP(), ov.first->cP()); });
401 std::set<size_t> kept{ov.second[0]};
402 for (bool grew = true; grew; ) // the nearest and what is joined to it along the polyline
403 {
404 grew = false;
405 for (size_t c : ov.second)
406 {
407 if (kept.count(c)) continue;
408 std::vector<VertexPointer> star;
409 edge::VVStarVE(&poly.vert[c], star);
410 for (VertexPointer w : star)
411 if (kept.count(tri::Index(poly, w))) { kept.insert(c); grew = true; break; }
412 }
413 }
414 for (size_t c : ov.second)
415 {
416 if (kept.count(c) || poly.vert[c].IsS()) continue;
417 // Only onto the nearest edge, if that snap alone is allowed.
418 ++deniedCnt;
419 CoordType &q = ip[c];
420 q = raw[c];
421 const int k = (q[0] <= q[1] && q[0] <= q[2]) ? 0 : (q[1] <= q[2] ? 1 : 2);
422 const FaceType &f = *face[c];
423 const ScalarType len = Distance(f.cP((k + 1) % 3), f.cP((k + 2) % 3));
424 if (q[k] <= par.barycentricSnapThr && len > 0 && q[k] * (DoubleArea(f) / len) <= par.maxSnapThr)
425 {
426 q[k] = 0;
427 q[(k + 1) % 3] /= q[(k + 1) % 3] + q[(k + 2) % 3];
428 q[(k + 2) % 3] = 1 - q[(k + 1) % 3];
429 }
430 }
431 }
432 int vertSnapCnt=0, edgeSnapCnt=0;
433 for (size_t i = 0; i < poly.vert.size(); ++i)
434 {
435 if (poly.vert[i].IsS()) continue; // control points stay where they are
436 int zeros = 0;
437 for (int k = 0; k < 3; ++k) zeros += ip[i][k] == 0;
438 if (zeros == 0) continue;
439 const FaceType &f = *face[i];
440 poly.vert[i].P() = f.cP(0)*ip[i][0] + f.cP(1)*ip[i][1] + f.cP(2)*ip[i][2];
441 if (zeros == 2) vertSnapCnt++; else edgeSnapCnt++;
442 }
443 const int dupCnt = CollapseNullEdges(poly);
444 Log("SnapPolyline %i vertices: snapped %i onto vertices and %i onto edges, %i kept off an occupied vertex, %i null edges collapsed",
445 poly.vn, vertSnapCnt, edgeSnapCnt, deniedCnt, dupCnt);
446 return vertSnapCnt==0 && edgeSnapCnt==0 && dupCnt==0;
447 }

513 /*
514 * Make an edge mesh 1-manifold by splitting all the
515 * vertexes that have more than two incident edges
516 *
517 * It performs the split in three steps.
518 * - First it collects and counts the vertices to be splitten.
519 * - Then it adds the vertices to the mesh and
520 * - lastly it updates the poly with the newly added vertices.
521 *
522 * singSplitFlag allows to ubersplit each singularity in a number of vertex of the same order of its degree.
523 * This is not really necessary but helps the management of sharp turns in the poly mesh.
524 * \todo add corner detection and split.
525 */
526
527 void DecomposeNonManifoldPolyline(MeshType &poly, bool singSplitFlag = true)
528 {
529 tri::Allocator<MeshType>::CompactEveryVector(poly);
530 std::vector<int> degreeVec(poly.vn, 0);
531 tri::UpdateTopology<MeshType>::VertexEdge(poly);
532 int neededVert=0;
533 int delta;
534 if(singSplitFlag) delta = 1;
535 else delta = 2;
536
537 for(VertexIterator vi=poly.vert.begin(); vi!=poly.vert.end();++vi)
538 {
539 std::vector<EdgeType *> starVec;
540 edge::VEStarVE(&*vi,starVec);
541 degreeVec[tri::Index(poly, *vi)] = starVec.size();
542 if(starVec.size()>2)
543 neededVert += starVec.size()-delta;
544 }
545 Log("DecomposeNonManifold Adding %i vert to a polyline of %i vert",neededVert,poly.vn);
546 VertexIterator firstVi = tri::Allocator<MeshType>::AddVertices(poly,neededVert);
547
548 for(size_t i=0;i<degreeVec.size();++i)
549 {
550 if(degreeVec[i]>2)
551 {
552 std::vector<EdgeType *> edgeStarVec;
553 edge::VEStarVE(&(poly.vert[i]),edgeStarVec);
554 assert(edgeStarVec.size() == degreeVec[i]);
555 for(size_t j=delta;j<edgeStarVec.size();++j)
556 {
557 EdgeType *ep = edgeStarVec[j];
558 int ind; // index of the vertex to be changed
559 if(tri::Index(poly,ep->V(0)) == i) ind = 0;
560 else ind = 1;
561
562 ep->V(ind) = &*firstVi;
563 ep->V(ind)->P() = poly.vert[i].P();
564 ep->V(ind)->N() = poly.vert[i].N();
565 ++firstVi;
566 }
567 }
568 }
569 assert(firstVi == poly.vert.end());
570 }

629 {
630 tri::UpdateTopology<MeshType>::VertexEdge(poly);
631 int cnt = 0;
632 for (bool changed = true; changed; )
633 {
634 changed = false;
635 for (size_t i = 0; i < poly.edge.size(); ++i)
636 {
637 EdgeType &e = poly.edge[i];
638 if (e.IsD() || e.V(0)->cP() != e.V(1)->cP()) continue;
639 if (e.V(0) == e.V(1)) { edge::VEDetach(e); tri::Allocator<MeshType>::DeleteEdge(poly, e); }
640 else edge::VEEdgeCollapseToVertex(poly, &e, e.V(1)->IsS() && !e.V(0)->IsS() ? 1 : 0); // keep a control point

642 changed = true;
643 }
644 }
645 tri::Allocator<MeshType>::CompactEveryVector(poly);
646 tri::UpdateTopology<MeshType>::VertexEdge(poly);
647 return cnt;

727 static bool BarycentricSnap(CoordType &ip, const FaceType &f, ScalarType barThr, ScalarType distThr)
728 {
729 // Setting coordinate i to 0 moves the point onto the opposite edge, by ip[i] times the
730 // height of the triangle over that edge.
731 const ScalarType area2 = DoubleArea(f);
732 bool zero[3];
733 for (int i = 0; i < 3; ++i)
734 {
735 const ScalarType len = Distance(f.cP((i + 1) % 3), f.cP((i + 2) % 3));
736 zero[i] = ip[i] <= barThr && len > 0 && ip[i] * (area2 / len) <= distThr;
737 }
738 if (zero[0] && zero[1] && zero[2]) // cannot leave the triangle: keep the largest
739 zero[ip[0] >= ip[1] && ip[0] >= ip[2] ? 0 : (ip[1] >= ip[2] ? 1 : 2)] = false;
740 ScalarType sum = 0;
741 for (int i = 0; i < 3; ++i) { if (zero[i]) ip[i] = 0; sum += ip[i]; }
742 int last = -1;
743 for (int i = 0; i < 3; ++i) if (!zero[i]) { ip[i] /= sum; last = i; }
744 // Make the sum exactly 1: the last kept coordinate takes what the others leave.
745 ip[last] = 1;
746 for (int i = 0; i < 3; ++i) if (i != last) ip[last] -= ip[i];
747 return zero[0] || zero[1] || zero[2];
748 }

751 // Given a segment find the maximum distance from it to the original surface.
752 // It is used to evaluate the Haustdorff distance of a Segment from the mesh.
753 ScalarType MaxSegDist(VertexType *v0, VertexType *v1, CoordType &farthestPointOnSurf, CoordType &farthestN, Distribution<ScalarType> *distanceDistribution=0)
754 {
755 ScalarType maxSurfDist = 0;
756 const ScalarType sampleNum = 10;
757 for(ScalarType k = 1;k<sampleNum;++k)
758 {
759 ScalarType surfDist;
760 CoordType closestPSurf;
761 CoordType samplePnt = (v0->P()*k +v1->P()*(sampleNum-k))/sampleNum;
762 FaceType *f = Closest(samplePnt, closestPSurf, surfDist);
763 if(distanceDistribution)
764 distanceDistribution->Add(surfDist);
765 if(surfDist > maxSurfDist)
766 {
767 maxSurfDist = surfDist;
768 farthestPointOnSurf = closestPSurf;
769 farthestN = f->N();
770 }
771 }
772 return maxSurfDist;
773 }

825 void RefineCurveByBaseMesh(MeshType &poly)
826 {
827 SnapPolyline(poly); // compacts poly
828 const int startEn = poly.en;
829 MeshType out;
830 tri::Allocator<MeshType>::AddVertices(out, poly.vert.size());
831 for (size_t i = 0; i < poly.vert.size(); ++i) out.vert[i].ImportData(poly.vert[i]);
832 for (size_t ei = 0; ei < poly.edge.size(); ++ei)
833 {
834 Progress(int(100 * ei / poly.edge.size()), "RefineCurveByBaseMesh: tracing segment %lu of %lu", (unsigned long)ei + 1, (unsigned long)poly.edge.size());

837 TraceSegment(poly.vert[i0].cP(), poly.vert[i1].cP(), crossings);
838 size_t prev = i0;
839 for (const CoordType &c : crossings)
840 {
841 const size_t vi = tri::Allocator<MeshType>::AddVertices(out, 1) - out.vert.begin();
842 out.vert[vi].ImportData(poly.vert[i0]);
843 out.vert[vi].P() = c;
844 out.vert[vi].ClearS(); // a crossing point is not one of the locked vertices
845 auto e = tri::Allocator<MeshType>::AddEdges(out, 1);
846 e->V(0) = &out.vert[prev]; e->V(1) = &out.vert[vi];
847 prev = vi;
848 }
849 auto e = tri::Allocator<MeshType>::AddEdges(out, 1);
850 e->V(0) = &out.vert[prev]; e->V(1) = &out.vert[i1];
851 }
852 poly.Clear();
853 tri::Append<MeshType, MeshType>::MeshCopy(poly, out);
854 // A geodesic is straight inside a face and along an edge: drop the free vertices that
855 // only bend a strand there, so it runs straight between its control points and edge
856 // crossings. A vertex is dropped when every face it belongs to also holds both its
857 // neighbours (inside a face: the neighbours are in it; on an edge: on the same edge),
858 // so the shortcut stays where the strand was; strands meeting the same edges in the
859 // same order cannot cross once straight.
860 tri::UpdateTopology<MeshType>::VertexEdge(poly);
861 int droppedCnt = 0;
862 auto faces = [&](const CoordType &p) {
863 std::vector<FaceType *> fs;
864 for (const auto &l : Locate(p)) fs.push_back(l.first);
865 std::sort(fs.begin(), fs.end());
866 return fs;
867 };
868 for (VertexType &v : poly.vert)
869 {
870 if (v.IsD() || v.IsS() || edge::VEDegree<EdgeType>(&v) != 2) continue;
871 std::vector<VertexPointer> nb;
872 edge::VVStarVE(&v, nb);
873 const std::vector<FaceType *> fv = faces(v.cP()), fa = faces(nb[0]->cP()), fb = faces(nb[1]->cP());
874 if (!std::includes(fa.begin(), fa.end(), fv.begin(), fv.end()) ||
875 !std::includes(fb.begin(), fb.end(), fv.begin(), fv.end())) continue;
876 edge::VEEdgeCollapse(poly, &v);
877 ++droppedCnt;
878 }
879 tri::Allocator<MeshType>::CompactEveryVector(poly);
880 SimplifyNullEdges(poly);
881 Log("RefineCurveByBaseMesh %i en -> %i en, %i vertices inside faces dropped", startEn, poly.en, droppedCnt);
882 }

905 {
906 Location loc{{f, ip}};
907 int zeros = 0, one = -1, zero = -1;
908 for (int i = 0; i < 3; ++i) { if (ip[i] == 0) { ++zeros; zero = i; } else one = i; }
909 if (zeros == 1) // on the edge opposite to corner `zero`: add the face across it

913 if (g != f) {
914 CoordType ig(0, 0, 0);
915 for (int j = 0; j < 3; ++j) {

917 if (g->V(j) == f->V((e + 1) % 3)) ig[j] = ip[(e + 1) % 3];
918 }
919 loc.push_back({g, ig});
920 }
921 }
922 else if (zeros == 2) // on vertex V(one): every face around it, across edges incident to it
923 {

925 for (size_t k = 0; k < loc.size(); ++k)
926 for (int i = 0; i < 3; ++i) {

930 bool seen = false;
931 for (const auto &l : loc) seen |= l.first == g;
932 if (seen) continue;
933 CoordType ig(0, 0, 0);
934 for (int j = 0; j < 3; ++j) if (g->V(j) == v) ig[j] = 1;
935 loc.push_back({g, ig});
936 }
937 }

942 void TraceSegment(const CoordType &p, const CoordType &q, std::vector<CoordType> &crossings)
943 {
944 const Location target = Locate(q);
945 Location cur = Locate(p);
946 for (int step = 0; ; ++step)
947 {
948 for (const auto &c : cur) for (const auto &t : target)
949 if (c.first == t.first) return; // both ends in one face: the rest lies in it
950 if (step > 4 * int(base.fn) + 16)
951 throw vcg::MissingPreconditionException("CoM: could not trace a curve segment across the mesh.");
952 // Among the faces of the current point, the one the segment heads into.
953 FaceType *g = nullptr;
954 CoordType bx, d;
955 ScalarType bestInward = -1;
956 for (const auto &c : cur)
957 {
958 CoordType bq;
959 InterpolationParameters(*c.first, c.first->N(), q, bq);
960 const CoordType dir = bq - c.second;
961 ScalarType inward = std::numeric_limits<ScalarType>::max();
962 for (int k = 0; k < 3; ++k) if (c.second[k] == 0) inward = std::min(inward, dir[k]);
963 if (inward > 0 && inward > bestInward) { bestInward = inward; g = c.first; bx = c.second; d = dir; }
964 }
965 if (g == nullptr)
966 throw vcg::MissingPreconditionException("CoM: a curve segment leaves the surface across a border.");
967 // Leave g where the first coordinate reaches zero.
968 int exitK = -1;
969 ScalarType sExit = std::numeric_limits<ScalarType>::max();
970 for (int k = 0; k < 3; ++k)
971 if (d[k] < 0 && bx[k] > 0 && bx[k] / -d[k] < sExit) { sExit = bx[k] / -d[k]; exitK = k; }
972 if (exitK < 0)
973 throw vcg::MissingPreconditionException("CoM: could not trace a curve segment across the mesh.");
974 CoordType b = bx + d * sExit;
975 b[exitK] = 0;
976 const ScalarType rest = b[(exitK + 1) % 3] + b[(exitK + 2) % 3];
977 b[(exitK + 1) % 3] /= rest;
978 b[(exitK + 2) % 3] = 1 - b[(exitK + 1) % 3];
979 RoundingSnap(b, *g); // through a vertex, if it passes that close to one
980 crossings.push_back(g->cP(0) * b[0] + g->cP(1) * b[1] + g->cP(2) * b[2]);
981 cur = LocationFrom(g, b);
982 }
983 }

1035 {

1037 std::vector<CoordType> smoothPosVec(poly.vn, CoordType(0, 0, 0));
1038 std::vector<int> cntVec(poly.vn, 0);

1045 }

1047 smoothPosVec[i] = (poly.vert[i].P() + smoothPosVec[i]) / ScalarType(cntVec[i] + 1);

1050 std::vector<CoordType> fieldPosVec(poly.vn, CoordType(0, 0, 0));
1051 for (const VertexType &v : poly.vert)

1054 FacePointer f = com.GetClosestFaceIP(v.P(),ip); // throws rather than return null
1055 ScalarType q = f->V(0)->Q() * ip[0] + f->V(1)->Q() * ip[1] + f->V(2)->Q() * ip[2];

1057

1059 }
1060 std::vector<CoordType> PosVec(poly.vn, CoordType(0, 0, 0));
1061 for (int i = 0; i < poly.vn; ++i)
1062 PosVec[i] = (smoothPosVec[i] * smoothBlend + fieldPosVec[i] * (1.0 - smoothBlend));
1063
1064 return PosVec;
1065 }

1273 static void SplitMeshWithPolyline(CoMType &com, MeshType &poly)
1274 {
1275 MeshType &m = com.base;
1276 tri::Allocator<MeshType>::CompactEveryVector(poly);
1277 if (poly.vn == 0) return;
1278 typedef std::pair<size_t, size_t> EdgeKey; // mesh vertex indices, smaller first
1279 auto key = [](size_t a, size_t b) { return a < b ? EdgeKey(a, b) : EdgeKey(b, a); };
1280
1281 // 1. Where each polyline vertex is on the mesh: on a vertex, on an edge, or in a face.
1282 enum { OnVertex, OnEdge, InFace };
1283 struct Loc { int kind; size_t vert; EdgeKey edge; ScalarType t; size_t face[2]; int faceNum; CoordType ip; };
1284 std::vector<Loc> loc(poly.vert.size());
1285 for (size_t ci = 0; ci < poly.vert.size(); ++ci)
1286 {
1287 Loc &l = loc[ci];
1288 FaceType *f = com.GetClosestFaceIP(poly.vert[ci].cP(), l.ip);
1289 // Only rounding-level snapping here: SnapPolyline already put the vertices that may
1290 // be snapped exactly on their vertex or edge, and kept the others off on purpose.
1291 com.RoundingSnap(l.ip, *f);
1292 int z = -1, one = -1;
1293 for (int i = 0; i < 3; ++i) { if (l.ip[i] == 0) z = i; if (l.ip[i] == 1) one = i; }
1294 l.face[0] = tri::Index(m, f); l.faceNum = 1;
1295 if (one >= 0) { l.kind = OnVertex; l.vert = tri::Index(m, f->V(one)); }
1296 else if (z >= 0) {
1297 // On the edge opposite to corner z: from V(z+1), at parameter ip[z+2].
1298 l.kind = OnEdge;
1299 const size_t a = tri::Index(m, f->V((z + 1) % 3)), b = tri::Index(m, f->V((z + 2) % 3));
1300 l.edge = key(a, b);
1301 l.t = (a < b) ? l.ip[(z + 2) % 3] : l.ip[(z + 1) % 3];
1302 const int e = (z + 1) % 3; // index of that edge in f
1303 if (f->FFp(e) != f) { l.face[1] = tri::Index(m, f->FFp(e)); l.faceNum = 2; }
1304 }
1305 else l.kind = InFace;
1306 }
1307
1308 // 2. A mesh vertex for each of them: the existing one, or a new one shared by every
1309 // polyline vertex within rounding distance (the mesh realizes the union of the
1310 // curves: two strands meeting at a point meet at one mesh vertex).
1311 const ScalarType tol = m.bbox.Diag() * ScalarType(1e-6);
1312 std::vector<size_t> meshV(poly.vert.size());
1313 std::map<EdgeKey, std::vector<std::pair<ScalarType, size_t>>> edgePoints; // (t from first, vertex)
1314 std::map<size_t, std::vector<std::pair<size_t, CoordType>>> facePoints; // (vertex, barycentric)
1315 size_t newVertNum = 0;
1316 const size_t firstNewVert = m.vert.size();
1317 for (size_t ci = 0; ci < poly.vert.size(); ++ci)
1318 {
1319 const Loc &l = loc[ci];
1320 if (l.kind == OnVertex) { meshV[ci] = l.vert; continue; }
1321 if (l.kind == OnEdge) {
1322 auto &pts = edgePoints[l.edge];
1323 const ScalarType len = Distance(m.vert[l.edge.first].cP(), m.vert[l.edge.second].cP());
1324 auto it = std::find_if(pts.begin(), pts.end(), [&](const std::pair<ScalarType, size_t> &p) { return std::abs(p.first - l.t) * len <= tol; });
1325 if (it != pts.end()) { meshV[ci] = it->second; continue; }
1326 meshV[ci] = firstNewVert + newVertNum++;
1327 pts.push_back({l.t, meshV[ci]});
1328 } else {
1329 auto &pts = facePoints[l.face[0]];
1330 const FaceType &f = m.face[l.face[0]];
1331 const CoordType p = f.cP(0) * l.ip[0] + f.cP(1) * l.ip[1] + f.cP(2) * l.ip[2];
1332 auto it = std::find_if(pts.begin(), pts.end(), [&](const std::pair<size_t, CoordType> &q) {
1333 return Distance(p, f.cP(0) * q.second[0] + f.cP(1) * q.second[1] + f.cP(2) * q.second[2]) <= tol; });
1334 if (it != pts.end()) { meshV[ci] = it->first; continue; }
1335 meshV[ci] = firstNewVert + newVertNum++;
1336 pts.push_back({meshV[ci], l.ip});
1337 }
1338 }
1339 tri::Allocator<MeshType>::AddVertices(m, newVertNum);
1340 for (size_t ci = 0; ci < poly.vert.size(); ++ci)
1341 {
1342 const Loc &l = loc[ci];
1343 VertexType &nv = m.vert[meshV[ci]];
1344 if (l.kind == OnEdge && meshV[ci] >= firstNewVert) {
1345 const VertexType &va = m.vert[l.edge.first], &vb = m.vert[l.edge.second];
1346 nv.P() = va.cP() * (1 - l.t) + vb.cP() * l.t;
1347 VertexInterpolator<MeshType>::Lerp(m, nv, va, vb, l.t);
1348 } else if (l.kind == InFace && meshV[ci] >= firstNewVert) {
1349 const FaceType &f = m.face[l.face[0]];
1350 nv.P() = f.cP(0) * l.ip[0] + f.cP(1) * l.ip[1] + f.cP(2) * l.ip[2];
1351 VertexInterpolator<MeshType>::Barycentric(m, nv, *f.cV(0), *f.cV(1), *f.cV(2), l.ip);
1352 }
1353 }
1354 for (size_t ci = 0; ci < poly.vert.size(); ++ci) poly.vert[ci].P() = m.vert[meshV[ci]].cP();
1355
1356 // 3. The segments: across one face, or along one edge.
1357 struct Segment { size_t polyEdge, a, b; }; // mesh vertices, in the polyline edge's direction
1358 std::map<size_t, std::vector<Segment>> faceSegments;
1359 std::vector<std::pair<Segment, EdgeKey>> edgeSegments;
1360 std::map<size_t, std::vector<size_t>> vertFaces; // faces around the polyline vertices on mesh vertices
1361 for (size_t ci = 0; ci < poly.vert.size(); ++ci)
1362 if (loc[ci].kind == OnVertex) vertFaces[loc[ci].vert];
1363 if (!vertFaces.empty())
1364 for (size_t fi = 0; fi < m.face.size(); ++fi) if (!m.face[fi].IsD())
1365 for (int i = 0; i < 3; ++i) {
1366 auto it = vertFaces.find(tri::Index(m, m.face[fi].V(i)));
1367 if (it != vertFaces.end()) it->second.push_back(fi);
1368 }
1369 auto candidateFaces = [&](size_t ci) {
1370 if (loc[ci].kind == OnVertex) return vertFaces[loc[ci].vert];
1371 return std::vector<size_t>(loc[ci].face, loc[ci].face + loc[ci].faceNum);
1372 };
1373 for (size_t ei = 0; ei < poly.edge.size(); ++ei)
1374 {
1375 const auto &e = poly.edge[ei];
1376 const size_t c0 = tri::Index(poly, e.cV(0)), c1 = tri::Index(poly, e.cV(1));
1377 if (meshV[c0] == meshV[c1]) continue; // a zero-length segment
1378 // The segment lies in the faces both its ends belong to: one face for a segment
1379 // across it, the two faces of an edge for a segment along it.
1380 std::vector<size_t> f0 = candidateFaces(c0), f1 = candidateFaces(c1), common;
1381 std::sort(f0.begin(), f0.end()); std::sort(f1.begin(), f1.end());
1382 std::set_intersection(f0.begin(), f0.end(), f1.begin(), f1.end(), std::back_inserter(common));
1383 if (common.empty())
1384 throw vcg::MissingPreconditionException("CoMEmbed: a curve segment does not lie in one face; call RefineCurveByBaseMesh first.");
1385 const Segment s{ei, meshV[c0], meshV[c1]};
1386 if (common.size() == 1) faceSegments[common[0]].push_back(s);
1387 else {
1388 const FaceType &f = m.face[common[0]]; // the edge both faces share
1389 EdgeKey ek(0, 0);
1390 for (int i = 0; i < 3; ++i)
1391 if (f.cFFp(i) == &m.face[common[1]]) ek = key(tri::Index(m, f.cV(i)), tri::Index(m, f.cV1(i)));
1392 edgeSegments.push_back({s, ek});
1393 }
1394 }
1395
1396 std::set<size_t> touched;
1397 for (const auto &fp : facePoints) touched.insert(fp.first);
1398 for (const auto &fs : faceSegments) touched.insert(fs.first);
1399 for (size_t fi = 0; fi < m.face.size(); ++fi) if (!m.face[fi].IsD())
1400 for (int i = 0; i < 3; ++i)
1401 if (edgePoints.count(key(tri::Index(m, m.face[fi].V(i)), tri::Index(m, m.face[fi].V1(i))))) touched.insert(fi);
1402
1403 // 4. Rebuild every touched face on its own. Where segments cross, the crossing point
1404 // becomes a mesh vertex; every segment is cut at the curve points lying on it
1405 // (crossings, other curves touching or overlapping it), and the pieces recovered.
1406 std::map<size_t, std::vector<size_t>> chains; // polyline edge -> mesh vertices inside it, in order
1407 std::vector<std::pair<size_t, CoordType>> crossingVerts; // (face, barycentric) of the new vertices
1408 const size_t firstCrossingVert = m.vert.size();
1409 std::vector<std::pair<size_t, LocalTriangulation>> rebuilt;
1410 size_t newFaceNum = 0, done = 0;
1411 for (size_t fi : touched)
1412 {
1413 com.Progress(int(100 * done / touched.size()), "SplitMeshWithPolyline: rebuilding face %lu of %lu", (unsigned long)done + 1, (unsigned long)touched.size());
1414 ++done;
1415 LocalTriangulation lt;
