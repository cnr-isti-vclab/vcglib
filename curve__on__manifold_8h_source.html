481 static CoordType QLerp(VertexType *v0, VertexType *v1)
482 {
483
484 ScalarType qSum = fabs(v0->Q())+fabs(v1->Q());
485 ScalarType w0 = (qSum - fabs(v0->Q()))/qSum;
486 ScalarType w1 = (qSum - fabs(v1->Q()))/qSum;
487 return v0->P()*w0 + v1->P()*w1;
488 }

503 bool SnapPolyline(MeshType &poly)
504 {
505 tri::Allocator<MeshType>::CompactEveryVector(poly);
506 tri::UpdateTopology<MeshType>::VertexEdge(poly);
507 int vertSnapCnt=0;
508 int edgeSnapCnt=0;
509 int borderCnt=0,midCnt=0,nonmanifCnt=0;
510 for(VertexIterator vi=poly.vert.begin(); vi!=poly.vert.end();++vi)
511 {
512 CoordType ip;
513 FaceType *f = GetClosestFaceIP(vi->cP(),ip);
514 if(BarycentricSnap(ip))
515 {
516 if(ip[0]>0 && ip[1]>0) { vi->P() = f->P(0)*ip[0]+f->P(1)*ip[1]; edgeSnapCnt++; assert(ip[2]==0); }
517 if(ip[0]>0 && ip[2]>0) { vi->P() = f->P(0)*ip[0]+f->P(2)*ip[2]; edgeSnapCnt++; assert(ip[1]==0); }
518 if(ip[1]>0 && ip[2]>0) { vi->P() = f->P(1)*ip[1]+f->P(2)*ip[2]; edgeSnapCnt++; assert(ip[0]==0); }
519
520 if(ip[0]==1.0) { vi->P() = f->P(0); vertSnapCnt++; assert(ip[1]==0 && ip[2]==0); }
521 if(ip[1]==1.0) { vi->P() = f->P(1); vertSnapCnt++; assert(ip[0]==0 && ip[2]==0); }
522 if(ip[2]==1.0) { vi->P() = f->P(2); vertSnapCnt++; assert(ip[0]==0 && ip[1]==0); }
523 }
524 else
525 {
526 int deg = edge::VEDegree<EdgeType>(&*vi);
527 if (deg > 2) nonmanifCnt++;
528 if (deg < 2) borderCnt++;
529 if (deg== 2) midCnt++;
530 }
531 }
532 Log("SnapPolyline %i vertices: snapped %i onto vert and %i onto edges %i nonmanif, %i border, %i mid",
533 poly.vn, vertSnapCnt, edgeSnapCnt, nonmanifCnt,borderCnt,midCnt);
534 int dupCnt=tri::Clean<MeshType>::RemoveDuplicateVertex(poly);
535 tri::Allocator<MeshType>::CompactEveryVector(poly);
536 if(dupCnt) Log("SnapPolyline: Removed %i Duplicated vertices",dupCnt);
537
538 return vertSnapCnt==0 && edgeSnapCnt==0 && dupCnt==0;
539 }

561 /*
562 * Make an edge mesh 1-manifold by splitting all the
563 * vertexes that have more than two incident edges
564 *
565 * It performs the split in three steps.
566 * - First it collects and counts the vertices to be splitten.
567 * - Then it adds the vertices to the mesh and
568 * - lastly it updates the poly with the newly added vertices.
569 *
570 * singSplitFlag allows to ubersplit each singularity in a number of vertex of the same order of its degree.
571 * This is not really necessary but helps the management of sharp turns in the poly mesh.
572 * \todo add corner detection and split.
573 */
574
575 void DecomposeNonManifoldPolyline(MeshType &poly, bool singSplitFlag = true)
576 {
577 tri::Allocator<MeshType>::CompactEveryVector(poly);
578 std::vector<int> degreeVec(poly.vn, 0);
579 tri::UpdateTopology<MeshType>::VertexEdge(poly);
580 int neededVert=0;
581 int delta;
582 if(singSplitFlag) delta = 1;
583 else delta = 2;
584
585 for(VertexIterator vi=poly.vert.begin(); vi!=poly.vert.end();++vi)
586 {
587 std::vector<EdgeType *> starVec;
588 edge::VEStarVE(&*vi,starVec);
589 degreeVec[tri::Index(poly, *vi)] = starVec.size();
590 if(starVec.size()>2)
591 neededVert += starVec.size()-delta;
592 }
593 Log("DecomposeNonManifold Adding %i vert to a polyline of %i vert",neededVert,poly.vn);
594 VertexIterator firstVi = tri::Allocator<MeshType>::AddVertices(poly,neededVert);
595
596 for(size_t i=0;i<degreeVec.size();++i)
597 {
598 if(degreeVec[i]>2)
599 {
600 std::vector<EdgeType *> edgeStarVec;
601 edge::VEStarVE(&(poly.vert[i]),edgeStarVec);
602 assert(edgeStarVec.size() == degreeVec[i]);
603 for(size_t j=delta;j<edgeStarVec.size();++j)
604 {
605 EdgeType *ep = edgeStarVec[j];
606 int ind; // index of the vertex to be changed
607 if(tri::Index(poly,ep->V(0)) == i) ind = 0;
608 else ind = 1;
609
610 ep->V(ind) = &*firstVi;
611 ep->V(ind)->P() = poly.vert[i].P();
612 ep->V(ind)->N() = poly.vert[i].N();
613 ++firstVi;
614 }
615 }
616 }
617 assert(firstVi == poly.vert.end());
618 }

667 void SimplifyMidEdge(MeshType &poly)
668 {
669 int startVn;
670 int midEdgeCollapseCnt=0;
671 tri::Allocator<MeshType>::CompactEveryVector(poly);
672 do
673 {
674 startVn = poly.vn;
675 for(int ei =0; ei<poly.en; ++ei)
676 {
677 VertexType *v0=poly.edge[ei].V(0);
678 VertexType *v1=poly.edge[ei].V(1);
679 CoordType ip0,ip1;
680 FaceType *f0=GetClosestFaceIP(v0->P(),ip0);
681 FaceType *f1=GetClosestFaceIP(v1->P(),ip1);
682
683 bool snap0=BarycentricSnap(ip0);
684 bool snap1=BarycentricSnap(ip1);
685 int e0i,e1i;
686 bool e0 = IsSnappedEdge(ip0,e0i);
687 bool e1 = IsSnappedEdge(ip1,e1i);
688 if(e0 && e1)
689 if( ( f0 == f1 && e0i == e1i) ||
690 ( f0 == f1->FFp(e1i) && e0i == f1->FFi(e1i)) ||
691 (f0->FFp(e0i) == f1 && f0->FFi(e0i) == e1i) ||
692 (f0->FFp(e0i) == f1->FFp(e1i) && f0->FFi(e0i) == f1->FFi(e1i)) )
693 {
694 CoordType newp = (v0->P()+v1->P())/2.0;
695 v0->P()=newp;
696 v1->P()=newp;
697 midEdgeCollapseCnt++;
698 }
699 }
700 tri::Clean<MeshType>::RemoveDuplicateVertex(poly);
701 tri::Allocator<MeshType>::CompactEveryVector(poly);
702// printf("SimplifyMidEdge %5i -> %5i %i mid %i ve \n",startVn,poly.vn,midEdgeCollapseCnt);
703 } while(startVn>poly.vn);
704 }

714 void SimplifyMidFace(MeshType &poly)
715 {
716 int startVn= poly.vn;;
717 int midFaceCollapseCnt=0;
718 int vertexEdgeCollapseCnt=0;
719 int curVn;
720 do
721 {
722 tri::Allocator<MeshType>::CompactEveryVector(poly);
723 curVn = poly.vn;
724 UpdateTopology<MeshType>::VertexEdge(poly);
725 for(int i =0; i<poly.vn;++i)
726 {
727 std::vector<VertexPointer> starVecVp;
728 edge::VVStarVE(&(poly.vert[i]),starVecVp);
729 if( (starVecVp.size()==2) )
730 {
731 CoordType ipP, ipN, ipI;
732 FacePointer fpP = GetClosestFaceIP(starVecVp[0]->P(),ipP);
733 FacePointer fpN = GetClosestFaceIP(starVecVp[1]->P(),ipN);
734 FacePointer fpI = GetClosestFaceIP(poly.vert[i].P(), ipI);
735
736 bool snapP = (BarycentricSnap(ipP));
737 bool snapN = (BarycentricSnap(ipN));
738 bool snapI = (BarycentricSnap(ipI));
739 VertexPointer vertexSnapP = 0;
740 VertexPointer vertexSnapN = 0;
741 VertexPointer vertexSnapI = 0;
742 for(int j=0;j<3;++j)
743 {
744 if(ipP[j]==1.0) vertexSnapP=fpP->V(j);
745 if(ipN[j]==1.0) vertexSnapN=fpN->V(j);
746 if(ipI[j]==1.0) vertexSnapI=fpI->V(j);
747 }
748
749 bool collapseFlag=false;
750
751 if((!snapI && snapP && snapN) || // First case a vertex that is not snapped between two snapped vertexes
752 (!snapI && !snapP && fpI==fpP) || // Or a two vertex not snapped but on the same face
753 (!snapI && !snapN && fpI==fpN) )
754 {
755 collapseFlag=true;
756 midFaceCollapseCnt++;
757 }
758
759 else // case 2) a vertex snap and edge snap we have to check that the edge do not share the same vertex of the vertex snap
760 if(snapI && snapP && snapN && vertexSnapI==0 && (vertexSnapP!=0 || vertexSnapN!=0) )
761 {
762 for(int j=0;j<3;++j) {
763 if(ipI[j]!=0 && (fpI->V(j)==vertexSnapP || fpI->V(j)==vertexSnapN)) {
764 collapseFlag=true;
765 vertexEdgeCollapseCnt++;
766 }
767 }
768 }
769
770 if(collapseFlag)
771 edge::VEEdgeCollapse(poly,&(poly.vert[i]));
772 }
773 }
774 } while(curVn>poly.vn);
775 Log("SimplifyMidFace %5i -> %5i %i mid %i ve",startVn,poly.vn,midFaceCollapseCnt,vertexEdgeCollapseCnt);
776 }

779 {
780 int startEn = poly.en;
781 Distribution<ScalarType> hist;
782 for(int i =0; i<poly.en;++i)
783 hist.Add(edge::Length(poly.edge[i]));
784
785 UpdateTopology<MeshType>::VertexEdge(poly);
786
787 for(int i =0; i<poly.vn;++i)
788 {
789 std::vector<VertexPointer> starVecVp;
790 edge::VVStarVE(&(poly.vert[i]),starVecVp);
791 if ((starVecVp.size()==2) && (!poly.vert[i].IsS()))

794 Segment3Type seg(starVecVp[0]->P(),starVecVp[1]->P());

802 {

804 }
805 }
806 }
807 tri::UpdateTopology<MeshType>::TestVertexEdge(poly);
808 tri::Allocator<MeshType>::CompactEveryVector(poly);
809 tri::UpdateTopology<MeshType>::TestVertexEdge(poly);
810// printf("Simplify %5i -> %5i (total len %5.2f)\n",startEn,poly.en,hist.Sum());
811 }

814 {
815 dist.Clear();
816 tri::UpdateTopology<MeshType>::VertexEdge(poly);
817 tri::UpdateQuality<MeshType>::VertexConstant(poly,0);

828 poly.vert[i].Q()/=deg;
829 }
830 tri::UpdateColor<MeshType>::PerVertexQualityRamp(poly,0,dist.Max());
831 }

1528 static void SplitMeshWithPolyline(CoMType &com, MeshType &poly)
1529 {
1530 MeshType &m = com.base;
1531 tri::Allocator<MeshType>::CompactEveryVector(poly);
1532 if (poly.vn == 0) return;
1533 typedef std::pair<size_t, size_t> EdgeKey; // mesh vertex indices, smaller first
1534 auto key = [](size_t a, size_t b) { return a < b ? EdgeKey(a, b) : EdgeKey(b, a); };
1535
1536 // 1. Where each polyline vertex is on the mesh: on a vertex, on an edge, or in a face.
1537 enum { OnVertex, OnEdge, InFace };
1538 struct Loc { int kind; size_t vert; EdgeKey edge; ScalarType t; size_t face[2]; int faceNum; CoordType ip; };
1539 std::vector<Loc> loc(poly.vert.size());
1540 for (size_t ci = 0; ci < poly.vert.size(); ++ci)
1541 {
1542 Loc &l = loc[ci];
1543 FaceType *f = com.GetClosestFaceIP(poly.vert[ci].cP(), l.ip);
1544 com.BarycentricSnap(l.ip);
1545 int z = -1, one = -1;
1546 for (int i = 0; i < 3; ++i) { if (l.ip[i] == 0) z = i; if (l.ip[i] == 1) one = i; }
1547 l.face[0] = tri::Index(m, f); l.faceNum = 1;
1548 if (one >= 0) { l.kind = OnVertex; l.vert = tri::Index(m, f->V(one)); }
1549 else if (z >= 0) {
1550 // On the edge opposite to corner z: from V(z+1), at parameter ip[z+2].
1551 l.kind = OnEdge;
1552 const size_t a = tri::Index(m, f->V((z + 1) % 3)), b = tri::Index(m, f->V((z + 2) % 3));
1553 l.edge = key(a, b);
1554 l.t = (a < b) ? l.ip[(z + 2) % 3] : l.ip[(z + 1) % 3];
1555 const int e = (z + 1) % 3; // index of that edge in f
1556 if (f->FFp(e) != f) { l.face[1] = tri::Index(m, f->FFp(e)); l.faceNum = 2; }
1557 }
1558 else l.kind = InFace;
1559 }
1560
1561 // 2. A mesh vertex for each of them: the existing one, or a new one shared by every
1562 // polyline vertex at exactly the same place.
1563 std::vector<size_t> meshV(poly.vert.size());
1564 std::map<std::pair<EdgeKey, ScalarType>, size_t> edgePointVert;
1565 size_t newVertNum = 0;
1566 const size_t firstNewVert = m.vert.size();
1567 for (size_t ci = 0; ci < poly.vert.size(); ++ci)
1568 {
1569 const Loc &l = loc[ci];
1570 if (l.kind == OnVertex) meshV[ci] = l.vert;
1571 else if (l.kind == OnEdge) {
1572 auto ins = edgePointVert.insert({{l.edge, l.t}, firstNewVert + newVertNum});
1573 if (ins.second) ++newVertNum;
1574 meshV[ci] = ins.first->second;
1575 }
1576 else meshV[ci] = firstNewVert + newVertNum++;
1577 }
1578 tri::Allocator<MeshType>::AddVertices(m, newVertNum);
1579 for (size_t ci = 0; ci < poly.vert.size(); ++ci)
1580 {
1581 const Loc &l = loc[ci];
1582 VertexType &nv = m.vert[meshV[ci]];
1583 if (l.kind == OnEdge) {

1820 }

1823 throw vcg::MissingPreconditionException("CoMEmbed: could not recover a curve segment; it passes through another curve point.");
1824 }
